382:    def _apply_diagnosis_safety(
383:        self, candidate: Optional[str], conf: int,
384:        symptoms: List[str], answers: Dict[str, Any],
385:    ) -> Tuple[Optional[str], int]:
386:        if not candidate:
387:            return candidate, conf
388:        key  = candidate.lower().strip()
389:        rule = DIAGNOSIS_SAFETY_RULES.get(key)
390:        if not rule:
391:            return candidate, conf
392:        all_text = " ".join(symptoms).lower() + " " + " ".join(str(v).lower() for v in answers.values())
393:        if not any(req in all_text for req in rule["required_any"]):
394:            return None, 0
395:        found = sum(1 for s in rule["supporting"] if s in all_text)
396:        if found < rule["required_count"]:
397:            return candidate, min(conf, 35)
398:        return candidate, conf

404:    def _validate_answer(self, message: str, step: Dict[str, Any]) -> Tuple[Any, bool, str]:
405:        q_type = step.get("type", "text")
406:        key    = step.get("key", "")
407:        raw    = message.strip()
408:        ctx    = key.replace("_", " ")
410:        if not raw:
411:            return None, False, ctx
413:        corrected = TextNormalizer.spell_correct(raw)
415:        # ── Semantic understanding layer (runs BEFORE strict validation) ──
416:        # Converts natural human answers into structured medical values.
417:        # e.g. "leg pain" -> yes for body_pain, "today morning" -> duration
418:        inferred = infer_medical_meaning(raw, key, q_type)
419:        if inferred is not None:
420:            value, _note = inferred
421:            return value, True, ctx
423:        if q_type == "yesno":
424:            normalized_yn = TextNormalizer.normalize_yes_no(corrected)
425:            if normalized_yn == "Yes":
426:                return "yes", True, ctx
427:            if normalized_yn == "No":
428:                return "no", True, ctx
429:            low = corrected.lower()
430:            if any(w in low for w in YES_VARIANTS):
431:                return "yes", True, ctx
432:            if any(w in low for w in NO_VARIANTS):
433:                return "no", True, ctx
434:            # Only reject if clearly off-topic
435:            if TextNormalizer.is_context_mismatch(raw, "yes_no"):
436:                return raw, False, ctx
437:            return raw, False, ctx
439:        if q_type == "number":
440:            sev, is_valid = TextNormalizer.parse_severity(corrected)
441:            if is_valid:
442:                return sev, True, ctx
443:            return raw, False, ctx
445:        if key == "temperature":
446:            if TextNormalizer.is_context_mismatch(raw, "temperature"):
447:                return raw, False, ctx
448:            tl = corrected.lower()
449:            if re.search(r"\d", corrected):
450:                return corrected, True, ctx
451:            if any(w in tl for w in ["not check", "not measure", "don't know",
452:                                     "unknown", "no thermometer", "didn't check",
453:                                     "normal", "high", "low", "slight", "mild"]):
454:                return corrected, True, ctx
455:            return raw, False, ctx
457:        if key == "duration":
458:            # Natural phrases already handled by infer_medical_meaning above.
459:            dur, is_valid = TextNormalizer.parse_duration(corrected)
460:            if is_valid:
461:                return f"{dur['value']} {dur['unit']}", True, ctx
462:            tl = corrected.lower()
463:            if re.search(
464:                r"(since|from|last|this)\s+(morning|evening|night|week|month|yesterday|today)|"
465:                r"(few|couple|several|a\s+few)\s+(hour|day|week|month)",
466:                tl
467:            ):

469:            return raw, False, ctx

485:                return corrected, True, ctx

488:        if len(raw) < 2:
489:            return raw, False, ctx
490:        return raw, True, ctx
